let liveTimeout=null; async function showSuggestLive(val){ let box=document.getElementById('suggestBox'); if(!val || val.trim().length<1){ box.style.display='none'; return; } let q=val.trim(); let qlow=q.toLowerCase(); // For 1-2 letters like "th" - use LOCAL list only (not live API - live API gives garbage "Th" for 2 letters) if(q.length < 3){ let local=LOCAL_PLACES.filter(p=>p.toLowerCase().startsWith(qlow)).sort((a,b)=>a.localeCompare(b)).slice(0,15); if(local.length===0) local=LOCAL_PLACES.filter(p=>p.toLowerCase().includes(qlow)).slice(0,15); box.innerHTML=local.map(r=>{ let highlighted=r.replace(new RegExp(`(${q})`,'i'), '$1'); let parts=r.split(' '); let country=parts[parts.length-1]||"World"; return `
${highlighted}${country}
`; }).join(''); box.style.display='block'; return; } // For 3+ letters like "tha", "mal" - use LIVE worldwide search box.innerHTML=`
🔍 Searching live worldwide for "${q}"...
`; box.style.display='block'; clearTimeout(liveTimeout); liveTimeout=setTimeout(async()=>{ try{ let url=`https://photon.komoot.io/api/?q=${encodeURIComponent(q)}&limit=15&lang=en`; let res=await fetch(url); let data=await res.json(); if(!data.features || data.features.length===0) throw new Error("no live"); let results=data.features.map(f=>{ let p=f.properties; let name=p.name||""; let city=p.city||p.county||p.state||""; let country=p.country||""; if(!name || name.length < 3) return null; let full=name + (city && city.toLowerCase()!==name.toLowerCase() ? `, ${city}` : "") + (country ? `, ${country}` : ""); return {full: full.trim(), name: name, country: country, city: city}; }).filter(r=>r && r.name && r.name.length>=3 && r.full.length>=4); let seen=new Set(); let unique=[]; for(let r of results){ let key=r.full.toLowerCase(); if(!seen.has(key)){ seen.add(key); unique.push(r); } if(unique.length>=15) break; } if(unique.length===0) throw new Error("empty"); box.innerHTML=unique.map(r=>{ let highlighted=r.full.replace(new RegExp(`(${q})`,'i'), '$1'); return `
${highlighted}${r.country||r.city||"World"}
`; }).join('') + `
Live from OpenStreetMap • Every village worldwide
`; box.style.display='block'; }catch(e){ let local=LOCAL_PLACES.filter(p=>p.toLowerCase().startsWith(qlow)).slice(0,15); if(local.length===0) local=LOCAL_PLACES.filter(p=>p.toLowerCase().includes(qlow)).slice(0,15); box.innerHTML=local.map(r=>{ let highlighted=r.replace(new RegExp(`(${q})`,'i'), '$1'); return `
${highlighted}India
`; }).join(''); box.style.display='block'; } }, 400); }
